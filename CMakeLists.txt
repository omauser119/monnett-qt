cmake_minimum_required(VERSION 3.24)
project(MonnettQt VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
include(GNUInstallDirs)
include(CTest)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Network Multimedia MultimediaWidgets)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SECRET REQUIRED IMPORTED_TARGET libsecret-1)
find_package(QuotientQt6 0.9.6 REQUIRED)
add_library(monnett_core STATIC src/json.cpp src/accounts.cpp src/api.cpp src/catalogue.cpp)
target_include_directories(monnett_core PUBLIC src)
target_link_libraries(monnett_core PUBLIC Qt6::Core Qt6::Gui Qt6::Network PkgConfig::SECRET)
target_compile_options(monnett_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(monnett-qt src/main.cpp src/window.cpp src/widgets.cpp src/authdialog.cpp src/operationpanel.cpp src/messages.cpp resources/resources.qrc)
target_link_libraries(monnett-qt PRIVATE monnett_core Qt6::Widgets Qt6::Multimedia Qt6::MultimediaWidgets QuotientQt6)
target_compile_options(monnett-qt PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS monnett-qt RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/social.monnett.MonnettQt.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES resources/monnett.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps RENAME social.monnett.MonnettQt.svg)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(monnett-tests tests/core_test.cpp resources/resources.qrc)
  target_link_libraries(monnett-tests PRIVATE monnett_core Qt6::Test)
  add_test(NAME core COMMAND monnett-tests)
  set_tests_properties(core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
